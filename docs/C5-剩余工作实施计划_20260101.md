1. **定义驱逐策略接口**

```go
type EvictionPolicy string

const (
    EvictionPolicyLRU  EvictionPolicy = "lru"  // Least Recently Used
    EvictionPolicyLFU  EvictionPolicy = "lfu"  // Least Frequently Used
    EvictionPolicyFIFO EvictionPolicy = "fifo" // First In First Out
    EvictionPolicyTTL  EvictionPolicy = "ttl"  // Time To Live
)

type EvictionStrategy interface {
    ShouldEvict(entry *CacheEntry, cache *EnhancedCache) bool
    SelectVictim(cache *EnhancedCache) string
}
```

2. **实现 LFU 策略**
```go
type LFUStrategy struct {
    frequencyMap map[string]int64
}

func (s *LFUStrategy) ShouldEvict(entry *CacheEntry, cache *EnhancedCache) bool {
    return len(cache.items) >= cache.maxSize
}

func (s *LFUStrategy) SelectVictim(cache *EnhancedCache) string {
    // 选择访问频率最低的条目
    var minFreq int64 = math.MaxInt64
    var victimKey string

    for key, element := range cache.items {
        entry := element.Value.(*CacheEntry)
        if entry.AccessCount < minFreq {
            minFreq = entry.AccessCount
            victimKey = key
        }
    }

    return victimKey
}
```

3. **实现 FIFO 策略**
```go
type FIFOStrategy struct {
    queue *list.List
}

func (s *FIFOStrategy) ShouldEvict(entry *CacheEntry, cache *EnhancedCache) bool {
    return len(cache.items) >= cache.maxSize
}

func (s *FIFOStrategy) SelectVictim(cache *EnhancedCache) string {
    if element := s.queue.Back(); element != nil {
        return element.Value.(string)
    }
    return ""
}
```

4. **集成策略到 EnhancedCache**
```go
type EnhancedCache struct {
    // ... 现有字段
    evictionPolicy EvictionStrategy
    policyType     EvictionPolicy
}

func (c *EnhancedCache) evict() {
    if c.evictionPolicy.ShouldEvict(nil, c) {
        victimKey := c.evictionPolicy.SelectVictim(c)
        c.Delete(victimKey)
    }
}
```

```go
func (w *CacheWarmer) WarmUp(ctx context.Context, cache *EnhancedCache) error {
    for _, key := range w.config.Keys {
        value, err := w.dataSource.Get(ctx, key)
        if err != nil {
            logger.Warnf("预热失败: %s, 错误: %v", key, err)

        }

    return nil
}
```

#### 2.1.3 实现缓存分片

**文件**: `pkg/ai/sharded_cache.go`（新文件）

**实施步骤**:

1. **定义分片缓存**
```go
type ShardedCache struct {
    shards   []*EnhancedCache
    numShards int
    hashFunc func(string) uint32
}

func NewShardedCache(ttl time.Duration, maxSize int, numShards int) *ShardedCache {
    shards := make([]*EnhancedCache, numShards)
    for i := 0; i < numShards; i++ {
        shards[i] = NewEnhancedCache(ttl, maxSize/numShards)
    }

    return &ShardedCache{
        shards:    shards,
        numShards: numShards,
        hashFunc:  fnv32a, // FNV-1a 32-bit 哈希
    }
}

func (sc *ShardedCache) getShard(key string) *EnhancedCache {
    hash := sc.hashFunc(key)
    return sc.shards[hash%uint32(sc.numShards)]
}

func (sc *ShardedCache) Get(key string) (interface{}, bool) {
    return sc.getShard(key).Get(key)
}

func (sc *ShardedCache) Set(key string, value interface{}) {
    sc.getShard(key).Set(key, value)
}

func (sc *ShardedCache) GetStats() []CacheStats {
    stats := make([]CacheStats, sc.numShards)
    for i, shard := range sc.shards {
        stats[i] = shard.GetStats()
    }
    return stats
}
```

2. **实现 FNV-1a 哈希函数**
```go
func fnv32a(key string) uint32 {
    const prime32 = uint32(16777619)
    hash := uint32(2166136261)
    for i := 0; i < len(key); i++ {
        hash ^= uint32(key[i])
        hash *= prime32
    }
    return hash
}
```

```markdown
# MMemory 性能调优指南

### 驱逐策略选择

- LFU: 适合热点数据稳定的场景
- FIFO: 适合简单 FIFO 队列场景

### 缓存大小调优
- 小型部署: 1000 条目
- 中型部署: 10000 条目
- 大型部署: 100000 条目

### 分片配置
- 并发访问 <100: 4 分片
- 并发访问 100-1000: 8 分片
- 并发访问 >1000: 16 分片

## 数据库配置
### 连接池调优
- MaxOpenConns: CPU 核心数 * 2
- MaxIdleConns: MaxOpenConns / 2
- ConnMaxLifetime: 5 分钟

### 查询优化
- 使用索引
- 避免全表扫描
- 使用批量操作
```
